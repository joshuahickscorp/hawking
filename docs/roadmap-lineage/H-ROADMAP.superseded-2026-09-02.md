fn hawking_campaign(root_goal: Goal, budget: Budget) {
    let authority = recover_disk_authority();
    let machine = census_machine_and_processes();
    let tools = tool_doctor();
    let providers = reconcile_providers_and_residents();
    let jobs = reconcile_background_jobs();

    assert_no_split_brain(authority);

    let mut frontier = compile_roadmap_frontier(
        root_goal,
        authority,
        machine,
        tools,
        providers,
        jobs,

    while budget.remaining() && !frontier.exhausted() {
        let wu = scheduler.select_highest_verified_value(frontier);
        let lease = resources.try_lease(wu.requirements);
        if lease.denied() {
            frontier.defer_or_backfill(wu);
            continue;
        }

        let evidence = context_compiler.compile(wu);
        let proposal = cognition.propose(wu, evidence);
        let mutation = tools.execute_if_authorized(proposal, wu.policy);

        genome.update(verdict);
        negative_science.update_if_needed(verdict);
        pareto.update_if_needed(verdict);

        frontier = frontier.reconcile_children(
            children_from_verified_evidence_only(wu, verdict)
        );

        handoff.maybe_checkpoint();
    }

    handoff.finalize_truthfully();
